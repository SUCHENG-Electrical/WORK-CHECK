        function generateOrderNo() {

            let nextNum = 1;

            if (workRecords.length > 0) {

                const nums = workRecords.map(r => {

                    const match = r.orderNo ? r.orderNo.match(/WR(\d{4})/) : null;

                    return match ? parseInt(match[1], 10) : 0;

                });

                const maxNum = Math.max(...nums, 0);

                nextNum = maxNum + 1;

            }

            if (nextNum > 9999) nextNum = 0;

            const formattedNo = `WR${String(nextNum).padStart(4, '0')}`;

            document.getElementById('display-order-no').innerText = `工作單號: ${formattedNo}`;

            return formattedNo;

        }

        function updateSubItems(selectElem, rowId) {

            const subSelect = row.querySelector('.subitem-select');

            

            subSelect.innerHTML = '<option value="">請選擇細項...</option>';

        function handleFormSubmit(event) {

            event.preventDefault();

            

            const orderNo = document.getElementById('display-order-no').innerText.replace("工作單號: ", "").trim();

            const region = document.getElementById('addr-region').value;

            const detailAddr = document.getElementById('addr-detail').value.trim();

            const phone = document.getElementById('cust-phone').value.trim();

            const name = document.getElementById('cust-name').value.trim();

            const staffTag = document.getElementById('staff-tag').value;

            const remark = document.getElementById('order-remark').value.trim();


            const rows = document.querySelectorAll('#items-container > div');

            const items = [];

            

            for (let row of rows) {

                const cat = row.querySelector('.category-select').value;

                const sub = row.querySelector('.subitem-select').value;

                const qty = row.querySelector('.qty-select').value;

                const note = row.querySelector('.item-note').value.trim();


                if (!cat || !sub) {

                    alert("請完整選擇所有項目的類別與細項！");

                    return;

                }

                items.push({ category: cat, subItem: sub, quantity: parseInt(qty), note: note });

            }


            const newRecord = {

                id: 'REC_' + Date.now(),

                orderNo: orderNo,

                date: new Date().toLocaleString('zh-HK'),

                region: region,

                address: detailAddr ? `${region} - ${detailAddr}` : region,

                phone: phone || '未提供',

                name: name || '未具名',

                staffTag: staffTag,

                items: items,

                remark: remark

            };


            workRecords.unshift(newRecord);

            localStorage.setItem(STORAGE_KEY, JSON.stringify(workRecords));


            alert(`工作單 ${orderNo} 儲存成功！`);

            resetForm();

            switchTab('records');

            renderRecordsList();

            updateStats();

        }

        }


        function renderRecordsList() {

            const tbody = document.getElementById('records-table-body');

            const keyword = document.getElementById('search-keyword').value.toLowerCase();

            const filterRegion = document.getElementById('filter-region').value;

            const filterTag = document.getElementById('filter-tag').value;


            tbody.innerHTML = '';


            const filtered = workRecords.filter(r => {

                const matchKeyword = (r.orderNo && r.orderNo.toLowerCase().includes(keyword)) || 

                                     (r.address && r.address.toLowerCase().includes(keyword)) || 

                                     (r.name && r.name.toLowerCase().includes(keyword)) || 

                                     (r.phone && r.phone.toLowerCase().includes(keyword));

                const matchRegion = filterRegion === "" || r.region === filterRegion;

                const matchTag = filterTag === "" || r.staffTag === filterTag;

                return matchKeyword && matchRegion && matchTag;

            });


            if (filtered.length === 0) {

                tbody.innerHTML = `<tr><td colspan="6" class="text-center py-8 text-slate-400">沒有找到相關工作紀錄</td></tr>`;

        function updateStats() {

            document.getElementById('stat-total-orders').innerText = workRecords.length;

            

            let totalItemsCount = 0;

            let catCounts = {};


            workRecords.forEach(r => {

                (r.items || []).forEach(i => {

                    totalItemsCount += i.quantity;

                    catCounts[i.category] = (catCounts[i.category] || 0) + i.quantity;

                });

            });


            document.getElementById('stat-total-items').innerText = totalItemsCount;


            let topCat = "無";

            let maxCount = 0;

            for (let cat in catCounts) {

                if (catCounts[cat] > maxCount) {

                    maxCount = catCounts[cat];

                    topCat = `${cat} (${maxCount}項)`;

                }

            }

            document.getElementById('stat-top-category').innerText = topCat;

        }
