        function populateSubItems(rowId, category, selectedSub = '') {

            const row = document.getElementById(rowId);

            if (!row) return;

            const subSelect = row.querySelector('.item-subitem');

            const items = CATEGORY_DATA[category] || [];


            subSelect.innerHTML = items.map(itm => {

                const isSel = itm === selectedSub ? 'selected' : '';

                return `<option value="${escapeQuote(itm)}" ${isSel}>${itm}</option>`;

            }).join('');

        }

            });


            if (items.length === 0) {

                alert('請選擇有效的工作項目細項！');

                return;

            }


            if (recordId) {

                const idx = records.findIndex(r => r.id === recordId);

                    id: 'REC-' + Date.now(),

                    region,

                    addressDetail,

                    phone,

                    customerName,

                    datetime,

                    status,

                    urgency,

                    remarks,

                    items

                };


                records.unshift(newRecord);

                incrementWOSeq(); // Increment counter WR0000 -> WR0001

                showToast('工作單 ' + woNo + ' 已成功開立！');

            }

            resetForm();

        function getStatusBadgeHTML(status) {

            if (status === '已完成') {

                return `<span class="px-2.5 py-1 rounded-full text-[11px] font-bold bg-emerald-100 text-emerald-700 inline-flex items-center gap-1"><i class="fa-solid fa-check"></i> 已完成</span>`;

            } else if (status === '進行中') {

                return `<span class="px-2.5 py-1 rounded-full text-[11px] font-bold bg-indigo-100 text-indigo-700 inline-flex items-center gap-1"><i class="fa-solid fa-spinner"></i> 進行中</span>`;

            } else if (status === '待跟進') {

                return `<span class="px-2.5 py-1 rounded-full text-[11px] font-bold bg-purple-100 text-purple-700 inline-flex items-center gap-1"><i class="fa-solid fa-arrows-rotate"></i> 待跟進</span>`;

            } else {

                return `<span class="px-2.5 py-1 rounded-full text-[11px] font-bold bg-amber-100 text-amber-700 inline-flex items-center gap-1"><i class="fa-solid fa-clock"></i> 未開始</span>`;

            }

        }

        function exportDataCSV() {

            if (records.length === 0) {

                alert('目前沒有紀錄可供匯出');

                return;

            }


            let csvContent = "data:text/csv;charset=utf-8,\uFEFF";

            csvContent += "工單編號,日期時間,對方名稱,電話,地區,詳細地址,工作狀態,緊急程度,單項工作內容清單,備註\n";


            records.forEach(r => {

                let itemsText = (r.items || []).map(i => `${i.category}:${i.subItem}(x${i.qty})`).join(' | ');


                const row = [

                    `"${r.woNo || ''}"`,

                    `"${r.datetime || ''}"`,

                    `"${r.customerName || ''}"`,

                    `"${r.phone || ''}"`,

                    `"${r.region || ''}"`,

                    `"${(r.addressDetail || '').replace(/"/g, '""')}"`,

                    `"${r.status || ''}"`,

                    `"${r.urgency || ''}"`,

                    `"${itemsText.replace(/"/g, '""')}"`,

                    `"${(r.remarks || '').replace(/"/g, '""')}"`

                ];

                csvContent += row.join(",") + "\n";

            });


            const encodedUri = encodeURI(csvContent);

            const link = document.createElement("a");

            link.setAttribute("href", encodedUri);

            link.setAttribute("download", `work_orders_${new Date().toISOString().slice(0,10)}.csv`);

            document.body.appendChild(link);

            link.click();

            link.remove();

        }
